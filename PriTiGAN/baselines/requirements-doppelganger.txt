# Separate venv for the DoppelGANger baseline only -- don't install into
# the main PriTiGAN env. ydata-synthetic switches TensorFlow into TF1
# graph mode process-wide, which breaks the rest of this repo.
#
#   python3.10 -m venv venv_doppelganger
#   source venv_doppelganger/bin/activate
#   pip install -r baselines/requirements-doppelganger.txt
#
# keras is pinned to 2.15.0 explicitly: if pip resolves keras>=3 alongside
# ydata-synthetic's tensorflow==2.15.* pin, it shadows TF 2.15's bundled
# tf.keras and breaks DoppelGANger's tf.compat.v1 code with a
# "initial_value arg to tf.Variable must be specified" error.

tensorflow==2.15.0
keras==2.15.0
ydata-synthetic>=1.3.0
pandas<3
numpy<2
